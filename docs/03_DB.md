```sql
PRAGMA foreign_keys = ON;
```

> ⚠️ `PRAGMA foreign_keys`는 DB 파일에 저장되지 않고 **그 연결에만** 적용됩니다. 백엔드는 DB 연결을 열 때마다 이 문장을 실행해야 합니다. 안 그러면 `REFERENCES` 검사와 `ON DELETE CASCADE / SET NULL`이 오류 없이 무시됩니다.

> 🕒 시각 컬럼(`created_at` 등)은 `datetime('now')` 기본값이라 **UTC** `YYYY-MM-DD HH:MM:SS`로 저장됩니다. API로 내보낼 때는 백엔드가 한국 시간(KST, +09:00)으로 바꿔서 API 명세의 형식(`YYYY-MM-DDTHH:mm`)으로 응답합니다. 고객이 보낸 `pickup_time`은 KST 기준 `YYYY-MM-DDTHH:mm` 그대로 저장하므로, `created_at`과 비교할 때는 둘 중 하나를 변환한 뒤 비교해야 합니다.

```sql
CREATE TABLE IF NOT EXISTS categories (
	category_id INTEGER PRIMARY KEY AUTOINCREMENT,
	name TEXT NOT NULL,
	parent_id INTEGER DEFAULT NULL REFERENCES categories(category_id) ON DELETE CASCADE,
	UNIQUE (parent_id, name)
);
```

이거는 뭐냐면 카테고리인데 트리구조라고 보면 됨<br>
예를 들어서<br>

```
한식
ㄴ 국밥
   ㄴ 돼지국밥
ㄴ 삼겹살
   ㄴ 솥뚜껑 삼겹살
중식
ㄴ 짜장면
```

이런식으로 카테고리가 구성이 됨<br>
이름은 **같은 부모 아래에서만** 안 겹치면 됨 (예: `한식 > 기타`, `중식 > 기타` 둘 다 가능)<br>
API에서 `category=한식`으로 필터링하면 `한식` 아래의 하위 카테고리(국밥, 돼지국밥 …)에 속한 가게까지 전부 포함함<br>
<br>

```sql
CREATE TABLE IF NOT EXISTS stores (
	store_id INTEGER PRIMARY KEY AUTOINCREMENT,
	owner_id INTEGER,
	name TEXT,
	category_id INTEGER REFERENCES categories(category_id) ON DELETE SET NULL,
	description TEXT,
	phone TEXT,
	address TEXT,
	location_lat REAL,
	location_lng REAL,
	open_hours TEXT,
	order_type TEXT DEFAULT 'none' CHECK (order_type IN ('preorder', 'reservation', 'none')),
	visits INTEGER DEFAULT 0,
	created_at DATETIME DEFAULT (datetime('now')) NOT NULL
);
```

이거는 가게 노드 <br>
`owner_id`는 이 가게 점주의 계정 id (로그인 계정 테이블의 `user_id`). 점주용 API는 로그인한 사용자의 id가 이 값과 같은지 확인함 <br>

```sql
CREATE TABLE IF NOT EXISTS coupons (
    coupon_id INTEGER PRIMARY KEY AUTOINCREMENT,

    store_id INTEGER NOT NULL
        REFERENCES stores(store_id)
        ON DELETE CASCADE,

    title TEXT NOT NULL,
    description TEXT,

    discount_rate REAL NOT NULL
        CHECK (discount_rate BETWEEN 0 AND 100),

    valid_from DATETIME,
    valid_until DATETIME,

    is_active INTEGER NOT NULL DEFAULT 1
        CHECK (is_active IN (0, 1)),

    created_at DATETIME NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS sign_keywords (
	id INTEGER PRIMARY KEY AUTOINCREMENT,
	name TEXT NOT NULL,
	store_id INTEGER NOT NULL REFERENCES stores(store_id) ON DELETE CASCADE,
	UNIQUE (store_id, name)
);

CREATE TABLE IF NOT EXISTS menus (
	menu_id INTEGER PRIMARY KEY AUTOINCREMENT,
	name TEXT NOT NULL,
	store_id INTEGER NOT NULL REFERENCES stores(store_id) ON DELETE CASCADE,
	price INTEGER NOT NULL,
	rating REAL CHECK (0 <= rating AND rating <= 5)
);

CREATE TABLE IF NOT EXISTS orders (
	order_id INTEGER PRIMARY KEY AUTOINCREMENT,
	store_id INTEGER REFERENCES stores(store_id) ON DELETE SET NULL,
	customer_id INTEGER NOT NULL,
	total_price INTEGER,
	pickup_time TEXT NOT NULL,
	customer_phone TEXT NOT NULL,
	status TEXT NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'accepted', 'done', 'rejected')),
	created_at DATETIME DEFAULT (datetime('now')) NOT NULL,
	is_active INTEGER DEFAULT 1
);

CREATE TABLE IF NOT EXISTS items (
	item_id INTEGER PRIMARY KEY AUTOINCREMENT,
	order_id INTEGER NOT NULL REFERENCES orders(order_id) ON DELETE CASCADE,
	menu_id INTEGER REFERENCES menus(menu_id) ON DELETE SET NULL,
	menu_name TEXT NOT NULL,
	quantity INTEGER DEFAULT 1 CHECK (quantity > 0),
	unit_price INTEGER NOT NULL CHECK (unit_price >= 0)
);

CREATE TABLE IF NOT EXISTS important_detail_updates (
	update_id INTEGER PRIMARY KEY AUTOINCREMENT,
	updated_table TEXT NOT NULL,
	updated_record_id INTEGER NOT NULL,
	updated_record_column TEXT NOT NULL,
	updated_before TEXT,
	updated_after TEXT,
	updated_at DATETIME DEFAULT (datetime('now')) NOT NULL
);
```

- `sign_keywords`는 `UNIQUE (store_id, name)`이 이미 `store_id`로 시작하는 인덱스를 만들어 주므로 `store_id` 인덱스를 따로 만들지 않음
- `orders.store_id`, `items.menu_id`는 `ON DELETE SET NULL`이라서 주문 기록이 있는 가게·메뉴도 지울 수 있음. 지워진 뒤에도 주문 내역이 보이도록 `items`에 주문 당시 메뉴 이름(`menu_name`)과 가격(`unit_price`)을 같이 저장함
- API의 주문 응답(`items`)은 이 `menu_name`, `unit_price`를 `name`, `unitPrice`로 내려줌. 메뉴가 지워졌으면 `menuId`만 `null`이 됨
- `orders.customer_id`는 주문한 고객의 계정 id (비회원 주문이면 비회원 세션 id). 고객은 자기 주문만 조회할 수 있음

```sql

CREATE INDEX IF NOT EXISTS idx_menus_store_id ON menus (store_id);

CREATE INDEX IF NOT EXISTS idx_stores_category_id ON stores (category_id);

CREATE INDEX IF NOT EXISTS idx_coupons_store_id ON coupons(store_id);

CREATE INDEX IF NOT EXISTS idx_stores_owner_id ON stores(owner_id);

CREATE INDEX IF NOT EXISTS idx_orders_store_id ON orders(store_id);

CREATE INDEX IF NOT EXISTS idx_orders_customer_id ON orders(customer_id);

CREATE INDEX IF NOT EXISTS idx_items_order_id ON items(order_id);

CREATE INDEX IF NOT EXISTS idx_items_menu_id ON items(menu_id);

CREATE INDEX IF NOT EXISTS idx_categories_parent_id ON categories(parent_id);

-- SQLite에서 NULL끼리는 UNIQUE 비교에서 서로 다른 값으로 취급되므로, 최상위 카테고리 이름 중복은 따로 막음
CREATE UNIQUE INDEX IF NOT EXISTS idx_categories_root_name ON categories(name) WHERE parent_id IS NULL;
```
